``` r
station_temperatures =
  noaa |>
  filter(month %in% c(1, 7), !is.na(tmax)) |>
```

## Import, clean data

``` r
bike_dir = file.path(tempdir(), "citibike")

unzip("data/citibike.zip", exdir = bike_dir)

bike_zips =
  list.files(
    file.path(bike_dir, "citibike"),
    pattern = "\\.zip$",
    full.names = TRUE
  )

for (file in bike_zips) {
  unzip(file, exdir = file.path(bike_dir, "citibike"))
}

bike =
  list.files(
    file.path(bike_dir, "citibike"),
    pattern = "\\.csv$",
    full.names = TRUE
  ) |>
  map_dfr(function(file) {
    read_csv(file, show_col_types = FALSE) |>
      mutate(
        year = as.integer(str_extract(basename(file), "20[0-9]{2}")),
        month = if_else(str_detect(basename(file), "^Jan"),
                        "January", "July")
      )
  }) |>
  mutate(
    month = factor(month, levels = c("January", "July")),
    weekdays = factor(
      weekdays,
      levels = c("Monday", "Tuesday", "Wednesday",
                 "Thursday", "Friday", "Saturday", "Sunday")
    ),
    member_casual = factor(
      member_casual,
      levels = c("member", "casual")
    ),
    rideable_type = factor(rideable_type)
  )

glimpse(bike)
```
